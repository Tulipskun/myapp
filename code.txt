package com.nte.fishing

import android.content.Context
import android.graphics.Bitmap
import android.graphics.PixelFormat
import android.hardware.display.DisplayManager
import android.hardware.display.VirtualDisplay
import android.media.ImageReader
import android.media.projection.MediaProjection
import android.util.DisplayMetrics
import kotlinx.coroutines.*
import org.json.JSONObject
import java.text.SimpleDateFormat
import java.util.*

data class IconPosition(val x: Int, val y: Int, val color: Int = 0)

class AutomationEngine(
    private val context: Context,
    private val metrics: DisplayMetrics,
    private val onLog: (String) -> Unit
) {
    private var job: Job? = null
    private val scope = CoroutineScope(Dispatchers.Default + SupervisorJob())

    var isRunning = false
        private set
    var isProjectionReady = false
        private set

    private var mediaProjection: MediaProjection? = null
    private var imageReader: ImageReader? = null
    private var virtualDisplay: VirtualDisplay? = null
    private val iconCache = mutableMapOf<String, IconPosition>()

    init { loadCache() }

    // ── Screen Capture ───────────────────────────────────────────────
    fun setupProjection(projection: MediaProjection) {
        mediaProjection = projection
        imageReader = ImageReader.newInstance(
            metrics.widthPixels, metrics.heightPixels, PixelFormat.RGBA_8888, 2
        )
        virtualDisplay = projection.createVirtualDisplay(
            "nte_capture",
            metrics.widthPixels, metrics.heightPixels, metrics.densityDpi,
            DisplayManager.VIRTUAL_DISPLAY_FLAG_AUTO_MIRROR,
            imageReader!!.surface, null, null
        )
        isProjectionReady = true
        log("📷 Screen capture พร้อมแล้ว")
    }

    fun captureScreen(): Bitmap? {
        if (!isProjectionReady) return null
        var image = imageReader?.acquireLatestImage()
        var retry = 0
        while (image == null && retry < 20) {
            Thread.sleep(100)
            image = imageReader?.acquireLatestImage()
            retry++
        }
        image ?: return null
        return try {
            val plane = image.planes[0]
            val buffer = plane.buffer
            buffer.rewind()
            val rowPadding = plane.rowStride - plane.pixelStride * image.width
            val bmp = Bitmap.createBitmap(
                image.width + rowPadding / plane.pixelStride,
                image.height, Bitmap.Config.ARGB_8888
            )
            bmp.copyPixelsFromBuffer(buffer)
            bmp
        } finally {
            image.close()
        }
    }

    // ── Region ───────────────────────────────────────────────────────
    fun scanRegion(left: Int, top: Int, right: Int, bottom: Int): Bitmap? {
        val full = captureScreen() ?: return null
        val l = left.coerceIn(0, full.width - 1)
        val t = top.coerceIn(0, full.height - 1)
        val r = right.coerceIn(l + 1, full.width)
        val b = bottom.coerceIn(t + 1, full.height)
        return try { Bitmap.createBitmap(full, l, t, r - l, b - t) }
        catch (e: Exception) { null }
    }

    fun getPixelColor(x: Int, y: Int): Int? {
        val bmp = captureScreen() ?: return null
        if (x >= bmp.width || y >= bmp.height) return null
        return bmp.getPixel(x, y)
    }

    // ── Orientation & Coordinate Transform ──────────────────────────
    private fun getRotation(): Int {
        val wm = context.getSystemService(Context.WINDOW_SERVICE) as android.view.WindowManager
        return if (android.os.Build.VERSION.SDK_INT >= 30)
            context.display?.rotation ?: 0
        else @Suppress("DEPRECATION") wm.defaultDisplay.rotation
    }

    fun displayW(): Int = metrics.widthPixels
    fun displayH(): Int = metrics.heightPixels

    // แปลง % → pixel realtime ตาม orientation ปัจจุบัน
    fun pctX(pct: Float): Int = (displayW() * pct).toInt()
    fun pctY(pct: Float): Int = (displayH() * pct).toInt()

    // แปลง game coord (landscape) → capture coord
    // ROTATION_90  (landscape ปกติ):  x_cap = y_game,          y_cap = W - x_game
    // ROTATION_270 (landscape กลับ):  x_cap = H - y_game,      y_cap = x_game
    fun toCaptureCoord(gx: Int, gy: Int): Pair<Int, Int> {
        return when (getRotation()) {
            android.view.Surface.ROTATION_90  -> Pair(gy, displayW() - gx)
            android.view.Surface.ROTATION_270 -> Pair(displayH() - gy, gx)
            else -> Pair(gx, gy)
        }
    }

    // ใช้แทน tap() ตอนเขียน logic — แปลง coord อัตโนมัติ
    fun tapGame(gx: Float, gy: Float) {
        val (cx, cy) = toCaptureCoord(gx.toInt(), gy.toInt())
        tap(cx.toFloat(), cy.toFloat())
    }

    // ── Icon Cache ───────────────────────────────────────────────────
    fun cacheIcon(name: String, x: Int, y: Int, color: Int? = null) {
        val c = color ?: getPixelColor(x, y) ?: 0
        iconCache[name] = IconPosition(x, y, c)
        saveCache()
        val hex = Integer.toHexString(c).uppercase().padStart(8, '0')
        log("📌 cache '$name' → ($x, $y) #$hex")
    }

    fun getCachedIcon(name: String): IconPosition? = iconCache[name]

    fun hasCachedIcons(vararg names: String): Boolean = names.all { iconCache.containsKey(it) }

    fun verifyIcon(name: String, tolerance: Int = 20): Boolean {
        val pos = iconCache[name] ?: return false
        val current = getPixelColor(pos.x, pos.y) ?: return false
        return colorDistance(pos.color, current) <= tolerance
    }

    fun colorDistance(c1: Int, c2: Int): Int {
        val r = Math.abs(((c1 shr 16) and 0xFF) - ((c2 shr 16) and 0xFF))
        val g = Math.abs(((c1 shr 8)  and 0xFF) - ((c2 shr 8)  and 0xFF))
        val b = Math.abs((c1 and 0xFF) - (c2 and 0xFF))
        return (r + g + b) / 3
    }

    fun clearCache() {
        iconCache.clear()
        context.getSharedPreferences("icon_cache", Context.MODE_PRIVATE)
            .edit().clear().apply()
        log("🗑️ ล้าง cache แล้ว")
    }

    private fun saveCache() {
        val json = JSONObject()
        iconCache.forEach { (name, pos) ->
            json.put(name, JSONObject().apply {
                put("x", pos.x)
                put("y", pos.y)
                put("color", pos.color)
            })
        }
        context.getSharedPreferences("icon_cache", Context.MODE_PRIVATE)
            .edit().putString("icons", json.toString()).apply()
    }

    private fun loadCache() {
        val str = context.getSharedPreferences("icon_cache", Context.MODE_PRIVATE)
            .getString("icons", null) ?: return
        try {
            val json = JSONObject(str)
            json.keys().forEach { name ->
                val obj = json.getJSONObject(name)
                iconCache[name] = IconPosition(
                    obj.getInt("x"),
                    obj.getInt("y"),
                    obj.optInt("color", 0)
                )
            }
            log("📦 โหลด cache: ${iconCache.size} icons")
        } catch (e: Exception) {
            log("⚠️ โหลด cache ไม่ได้: ${e.message}")
        }
    }

    // ── Gesture ──────────────────────────────────────────────────────
    fun tap(x: Float, y: Float) = FishingAccessibilityService.instance?.tap(x, y)

    fun tapIcon(name: String) {
        val pos = getCachedIcon(name)
        if (pos != null) tap(pos.x.toFloat(), pos.y.toFloat())
        else log("⚠️ ไม่พบ cache ของ '$name'")
    }

    fun longPress(x: Float, y: Float, duration: Long = 800) =
        FishingAccessibilityService.instance?.longPress(x, y, duration)

    fun swipe(x1: Float, y1: Float, x2: Float, y2: Float, duration: Long = 300) =
        FishingAccessibilityService.instance?.swipe(x1, y1, x2, y2, duration)

    // ── Log ──────────────────────────────────────────────────────────
    fun log(msg: String) {
        val time = SimpleDateFormat("HH:mm:ss", Locale.US).format(Date())
        onLog("[$time] $msg")
    }

    // ── Loop ─────────────────────────────────────────────────────────
    fun start() {
        if (isRunning) return
        isRunning = true
        log("🟢 เริ่ม Automation")
        job = scope.launch { loop() }
    }

    fun stop() {
        if (!isRunning) return
        job?.cancel()
        isRunning = false
        log("🔴 หยุด Automation")
    }

    private suspend fun loop() {
        while (isRunning) {
            if (!hasCachedIcons("cast", "status")) {
                log("🔍 กำลัง scan หา icons...")
                val offsetX = pctX(0.78f)
                val offsetY = pctY(0.85f)
                val bmp = scanRegion(offsetX, offsetY, pctX(0.98f), pctY(0.99f))
                if (bmp == null) {
                    log("⚠️ ถ่ายภาพไม่ได้")
                    delay(1000)
                    continue
                }

                // 1) หา bright pixels
                val threshold = 160
                val brightPixels = mutableListOf<Pair<Int, Int>>()
                for (y in 0 until bmp.height) {
                    for (x in 0 until bmp.width) {
                        val c = bmp.getPixel(x, y)
                        val r = (c shr 16) and 0xFF
                        val g = (c shr 8)  and 0xFF
                        val b =  c         and 0xFF
                        if (r > threshold && g > threshold && b > threshold)
                            brightPixels.add(Pair(x + offsetX, y + offsetY))
                    }
                }

                // 2) จัดกลุ่ม pixel ที่อยู่ใกล้กัน → clusters
                val clusterDist = 12
                val clusters = mutableListOf<MutableList<Pair<Int, Int>>>()
                for (p in brightPixels) {
                    val match = clusters.firstOrNull { cl ->
                        cl.any {
                            Math.abs(it.first - p.first) < clusterDist &&
                            Math.abs(it.second - p.second) < clusterDist
                        }
                    }
                    if (match != null) match.add(p) else clusters.add(mutableListOf(p))
                }

                // 3) filter เฉพาะ cluster ที่รูปทรงกลม (w ≈ h) และขนาดสมเหตุสมผล
                val circularIcons = clusters.mapNotNull { pts ->
                    val minX = pts.minOf { it.first };  val maxX = pts.maxOf { it.first }
                    val minY = pts.minOf { it.second }; val maxY = pts.maxOf { it.second }
                    val w = maxX - minX
                    val h = maxY - minY
                    val size = maxOf(w, h)
                    // ขนาด 10-90px และ aspect ratio > 0.45 (ไม่ยาวเกิน = ไม่ใช่แสงน้ำ)
                    if (size < 10 || size > 90) return@mapNotNull null
                    if (w == 0 || h == 0) return@mapNotNull null
                    if (minOf(w, h).toFloat() / maxOf(w, h) < 0.45f) return@mapNotNull null
                    // คืน center ของ cluster
                    Pair(
                        pts.sumOf { it.first } / pts.size,
                        pts.sumOf { it.second } / pts.size
                    )
                }.sortedByDescending { it.first }

                log("📊 bright=${brightPixels.size} clusters=${clusters.size} circular=${circularIcons.size} bmp=${bmp.width}x${bmp.height}")

                if (circularIcons.size >= 2) {
                    val (cx0, cy0) = toCaptureCoord(circularIcons[0].first, circularIcons[0].second)
                    val (cx1, cy1) = toCaptureCoord(circularIcons[1].first, circularIcons[1].second)
                    cacheIcon("cast",   cx0, cy0)
                    cacheIcon("status", cx1, cy1)
                    log("✅ cast($cx0, $cy0) status($cx1, $cy1)")
                } else {
                    log("⚠️ พบ circular icon แค่ ${circularIcons.size} ตัว ลองใหม่...")
                    delay(1000)
                    continue
                }
            }

            // TODO: logic ตกปลา
            delay(100)
        }
    }

    // ── Cleanup ──────────────────────────────────────────────────────
    fun destroy() {
        stop()
        virtualDisplay?.release()
        imageReader?.close()
        mediaProjection?.stop()
        scope.cancel()
    }
}

import android.accessibilityservice.AccessibilityService
import android.accessibilityservice.GestureDescription
import android.graphics.Path
import android.view.accessibility.AccessibilityEvent

class FishingAccessibilityService : AccessibilityService() {

    companion object {
        var instance: FishingAccessibilityService? = null
            private set

        fun isEnabled() = instance != null
    }

    override fun onServiceConnected() {
        super.onServiceConnected()
        instance = this
    }

    override fun onDestroy() {
        super.onDestroy()
        instance = null
    }

    override fun onAccessibilityEvent(event: AccessibilityEvent?) {}
    override fun onInterrupt() {}

    // แตะจุดเดียว
    fun tap(x: Float, y: Float) {
        val path = Path().apply { moveTo(x, y) }
        val stroke = GestureDescription.StrokeDescription(path, 0, 50)
        dispatchGesture(GestureDescription.Builder().addStroke(stroke).build(), null, null)
    }

    // กดค้าง
    fun longPress(x: Float, y: Float, duration: Long = 800) {
        val path = Path().apply { moveTo(x, y) }
        val stroke = GestureDescription.StrokeDescription(path, 0, duration)
        dispatchGesture(GestureDescription.Builder().addStroke(stroke).build(), null, null)
    }

    // ปัด
    fun swipe(x1: Float, y1: Float, x2: Float, y2: Float, duration: Long = 300) {
        val path = Path().apply { moveTo(x1, y1); lineTo(x2, y2) }
        val stroke = GestureDescription.StrokeDescription(path, 0, duration)
        dispatchGesture(GestureDescription.Builder().addStroke(stroke).build(), null, null)
    }
}

package com.nte.fishing

import android.app.*
import android.content.Intent

class FloatingService : Service() {
    private lateinit var wm: WindowManager
    private var iconView: View? = null
    private var menuView: View? = null
    private lateinit var engine: AutomationEngine
    private val logBuf = StringBuilder()

    override fun onBind(intent: Intent?): IBinder? = null

    override fun onCreate() {
        super.onCreate()
        wm = getSystemService(WINDOW_SERVICE) as WindowManager
        engine = AutomationEngine(this, resources.displayMetrics) { msg -> appendLog(msg) }
        startFg()
        showIcon()
        appendLog("🟡 Service เริ่มทำงานแล้ว")
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        appendLog("📩 onStartCommand ถูกเรียก")
        val code = intent?.getIntExtra("proj_code", -1) ?: -1
        val data = intent?.getParcelableExtra<Intent>("proj_data")
        appendLog("proj_code=$code  data=${data != null}")

        if (data != null) {
            Thread {
                try {
                    appendLog("1/3 กำลังสร้าง projection...")
                    val projection = (getSystemService(MEDIA_PROJECTION_SERVICE) as MediaProjectionManager)
                        .getMediaProjection(code, data)

                    // ✅ Android 14+ (API 34) บังคับให้ลงทะเบียน Callback ก่อนใช้งาน
                    projection.registerCallback(object : MediaProjection.Callback() {
                        override fun onStop() {
                            appendLog("🔴 MediaProjection หยุดโดยระบบ (ผู้ใช้ยกเลิก/แอปเข้าพื้นหลัง)")
                            // หมายเหตุ: หากต้องการรีเซ็ตสถานะ isProjectionReady 
                            // แนะนำให้เพิ่มเมธอด public fun resetProjection() ใน AutomationEngine                            super.onStop()
                        }
                    }, null)

                    appendLog("2/3 ได้ projection แล้ว")
                    engine.setupProjection(projection)
                    appendLog("3/3 ✅ Projection พร้อมแล้ว")
                } catch (e: Exception) {
                    appendLog("❌ Error: ${e.message}")
                    e.printStackTrace()
                }
            }.start()
        } else {
            appendLog("⚠️ ไม่มี projection data")
        }
        return START_NOT_STICKY
    }

    private fun startFg() {
        val ch = "nte_ch"
        if (Build.VERSION.SDK_INT >= 26) {
            val channel = NotificationChannel(ch, "NTE Fishing", NotificationManager.IMPORTANCE_LOW)
            (getSystemService(NOTIFICATION_SERVICE) as NotificationManager).createNotificationChannel(channel)
        }
        startForeground(1, NotificationCompat.Builder(this, ch)
            .setContentTitle("NTE Fishing Bot")
            .setContentText("กำลังทำงาน...")
            .setSmallIcon(android.R.drawable.ic_menu_compass)
            .build())
    }

    private fun showIcon() {
        iconView = LayoutInflater.from(this).inflate(R.layout.layout_floating_icon, null)
        val params = iconParams()
        var initX = 0; var initY = 0; var initTx = 0f; var initTy = 0f; var drag = false
        iconView!!.setOnTouchListener { _, e ->
            when (e.action) {
                MotionEvent.ACTION_DOWN -> { initX=params.x; initY=params.y; initTx=e.rawX; initTy=e.rawY; drag=false; false }
                MotionEvent.ACTION_MOVE -> {
                    if (Math.abs(e.rawX-initTx)>5||Math.abs(e.rawY-initTy)>5) {
                        drag=true; params.x=initX+(e.rawX-initTx).toInt(); params.y=initY+(e.rawY-initTy).toInt()
                        wm.updateViewLayout(iconView, params)
                    }; true
                }
                MotionEvent.ACTION_UP -> { if (!drag) { removeIcon(); showMenu() }; false }
                else -> false
            }
        }
        wm.addView(iconView, params)
    }
    private fun showMenu() {
        menuView = LayoutInflater.from(this).inflate(R.layout.layout_floating_menu, null)
        val params = WindowManager.LayoutParams(
            WindowManager.LayoutParams.WRAP_CONTENT,
            WindowManager.LayoutParams.WRAP_CONTENT,
            overlayType(),
            WindowManager.LayoutParams.FLAG_NOT_TOUCH_MODAL,
            PixelFormat.TRANSLUCENT
        ).apply { gravity = Gravity.BOTTOM or Gravity.CENTER_HORIZONTAL; y = 40 }

        val tvLog    = menuView!!.findViewById<TextView>(R.id.tvLog)
        val tvStatus = menuView!!.findViewById<TextView>(R.id.tvRunStatus)
        tvLog.text   = logBuf.toString().ifEmpty { "[รอเริ่มต้น...]" }

        menuView!!.findViewById<Button>(R.id.btnAutoStart).setOnClickListener {
            if (!FishingAccessibilityService.isEnabled()) {
                appendLog("⚠️ กรุณาเปิด Accessibility ก่อน")
                return@setOnClickListener
            }
            if (!engine.isProjectionReady) {
                appendLog("⚠️ Screen capture ยังไม่พร้อม")
                return@setOnClickListener
            }
            engine.start()
            tvStatus.text = "กำลังทำงาน"
            tvStatus.setTextColor(0xFF00C853.toInt())
        }
        menuView!!.findViewById<Button>(R.id.btnAutoStop).setOnClickListener {
            engine.stop()
            tvStatus.text = "หยุดทำงาน"
            tvStatus.setTextColor(0xFFFF6B6B.toInt())
        }
        menuView!!.findViewById<Button>(R.id.btnHide).setOnClickListener { removeMenu(); showIcon() }
        menuView!!.findViewById<Button>(R.id.btnReset).setOnClickListener { engine.clearCache() }
        menuView!!.findViewById<Button>(R.id.btnClose).setOnClickListener { engine.stop(); stopSelf() }
        wm.addView(menuView, params)
    }

    private fun removeIcon() { iconView?.let { wm.removeView(it) }; iconView = null }
    private fun removeMenu() { menuView?.let { wm.removeView(it) }; menuView = null }

    private fun appendLog(msg: String) {
        logBuf.appendLine(msg)
        val lines = logBuf.lines().takeLast(100)
        logBuf.clear()
        logBuf.append(lines.joinToString("\n"))
        menuView?.post {
            menuView?.findViewById<TextView>(R.id.tvLog)?.text = logBuf.toString()
            menuView?.findViewById<ScrollView>(R.id.scrollLog)?.post {
                menuView?.findViewById<ScrollView>(R.id.scrollLog)?.fullScroll(View.FOCUS_DOWN)            }
        }
    }

    private fun iconParams() = WindowManager.LayoutParams(
        WindowManager.LayoutParams.WRAP_CONTENT,
        WindowManager.LayoutParams.WRAP_CONTENT,
        overlayType(),
        WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE,
        PixelFormat.TRANSLUCENT
    ).apply { gravity = Gravity.TOP or Gravity.START; x = 0; y = 300 }

    private fun overlayType() =
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O)
            WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY
        else @Suppress("DEPRECATION") WindowManager.LayoutParams.TYPE_PHONE

    override fun onDestroy() {
        super.onDestroy()
        engine.destroy()
        removeIcon()
        removeMenu()
    }

    // ดึงไฟล์จาก GitHub API → คืน Pair(content, sha)
    private fun fetchFile(): Pair<String, String> {
        val conn = (URL(GITHUB_API_URL).openConnection() as HttpURLConnection).apply {

            val body = JSONObject().apply {
                put("message", message)
                put("content", encoded)
                put("sha", sha)
            }
            val conn = (URL(GITHUB_API_URL).openConnection() as HttpURLConnection).apply {
                requestMethod = "PUT"
                setRequestProperty("Authorization", "Bearer $GITHUB_TOKEN")
                setRequestProperty("Content-Type", "application/json")
                doOutput = true
            }
            conn.outputStream.write(body.toString().toByteArray())
            conn.responseCode in 200..299
        } catch (e: Exception) { false }
    }

    suspend fun validate(context: Context, key: String): KeyResult = withContext(Dispatchers.IO) {
        if (key.trim() == ADMIN_KEY)
            return@withContext KeyResult(true, "✓ Admin key")

        try {
            val (content, sha) = fetchFile()
            val root   = JSONObject(content)
            val arr    = root.getJSONArray("keys")
            val device = getDeviceId(context)
            val sdf    = SimpleDateFormat("yyyy-MM-dd", Locale.US)
            val today  = Date()

            for (i in 0 until arr.length()) {
                val obj = arr.getJSONObject(i)
                if (obj.getString("key") != key.trim()) continue

                if (!obj.getBoolean("active"))
                    return@withContext KeyResult(false, "✗ Key ถูกปิดใช้งาน")

                val expires = sdf.parse(obj.getString("expires")) ?: Date(0)
                if (expires.before(today))
                    return@withContext KeyResult(false, "✗ Key หมดอายุ")

                val boundDevice = obj.optString("device_id", "")

                // ยังไม่ผูกเครื่อง → ผูกอัตโนมัติ
                if (boundDevice.isEmpty()) {
                    obj.put("device_id", device)
                    val ok = pushFile(root.toString(2), sha, "bind: ${key.trim()} → $device")
                    return@withContext KeyResult(
                        true,
                        if (ok) "✓ Key ถูกต้อง — ผูกกับเครื่องนี้แล้ว"
                        else    "✓ Key ถูกต้อง (บันทึก HWID ไม่สำเร็จ)"
                    )
                }

                if (boundDevice != device)
                    return@withContext KeyResult(false, "✗ Key นี้ผูกกับอุปกรณ์อื่นแล้ว")

                return@withContext KeyResult(true, "✓ Key ถูกต้อง")
            }

            KeyResult(false, "✗ ไม่พบ Key นี้ในระบบ")

        } catch (e: Exception) {
            KeyResult(false, "✗ เชื่อมต่อไม่ได้: ${e.message}")
        }
    }
}

package com.nte.fishing

import android.content.Intent

import kotlinx.coroutines.launch

private const val PREFS       = "nte_prefs"
private const val KEY_LAST    = "last_key"
private const val REQ_OVERLAY = 1001
private const val REQ_PROJECT = 1002

class MainActivity : AppCompatActivity() {

    private lateinit var etKey: TextInputEditText
    private lateinit var tvStatus: TextView
    private lateinit var btnValidate: Button
    private lateinit var layoutActions: View
    private lateinit var progressBar: ProgressBar
    private lateinit var projectionManager: MediaProjectionManager

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main)

        etKey         = findViewById(R.id.etKey)
        tvStatus      = findViewById(R.id.tvStatus)
        btnValidate   = findViewById(R.id.btnValidate)
        layoutActions = findViewById(R.id.layoutActions)
        progressBar   = findViewById(R.id.progressBar)

        projectionManager = getSystemService(MEDIA_PROJECTION_SERVICE) as MediaProjectionManager

        getSharedPreferences(PREFS, MODE_PRIVATE)
            .getString(KEY_LAST, "")?.takeIf { it.isNotEmpty() }
            ?.let { etKey.setText(it) }

        btnValidate.setOnClickListener {
            val key = etKey.text.toString().trim()
            if (key.isEmpty()) { showStatus("กรุณาใส่ Key", false); return@setOnClickListener }
            doValidate(key)        }

        findViewById<Button>(R.id.btnStart).setOnClickListener {
            if (checkAccessibility()) {
                checkOverlay()
            }
        }

        findViewById<Button>(R.id.btnExit).setOnClickListener { finish() }
    }

    private fun checkAccessibility(): Boolean {
        if (!FishingAccessibilityService.isEnabled()) {
            Toast.makeText(this, "กรุณาเปิดสิทธิ์ Accessibility สำหรับ NTE Fishing", Toast.LENGTH_LONG).show()
            val intent = Intent(Settings.ACTION_ACCESSIBILITY_SETTINGS)
            startActivity(intent)
            return false
        }
        return true
    }

    private fun doValidate(key: String) {
        setLoading(true)
        lifecycleScope.launch {
            val result = KeyValidator.validate(this@MainActivity, key)
            setLoading(false)
            showStatus(result.message, result.valid)
            if (result.valid) {
                getSharedPreferences(PREFS, MODE_PRIVATE).edit()
                    .putString(KEY_LAST, key).apply()
                layoutActions.visibility = View.VISIBLE
            }
        }
    }

    private fun checkOverlay() {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M && !Settings.canDrawOverlays(this)) {
            startActivityForResult(
                Intent(Settings.ACTION_MANAGE_OVERLAY_PERMISSION, Uri.parse("package:$packageName")),
                REQ_OVERLAY
            )
        } else {
            checkProjection()
        }
    }

    private fun checkProjection() {
        startActivityForResult(projectionManager.createScreenCaptureIntent(), REQ_PROJECT)
    }
    @Deprecated("onActivityResult")
    override fun onActivityResult(requestCode: Int, resultCode: Int, data: Intent?) {
        super.onActivityResult(requestCode, resultCode, data)
        when (requestCode) {
            REQ_OVERLAY -> {
                if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M && Settings.canDrawOverlays(this))
                    checkProjection()
                else
                    Toast.makeText(this, "ต้องอนุญาต Overlay", Toast.LENGTH_SHORT).show()
            }
            REQ_PROJECT -> {
                if (resultCode == RESULT_OK && data != null)
                    launchService(resultCode, data)
                else
                    Toast.makeText(this, "ต้องอนุญาต Screen Capture", Toast.LENGTH_SHORT).show()
            }
        }
    }

    private fun launchService(code: Int, data: Intent) {
        val intent = Intent(this, FloatingService::class.java).apply {
            putExtra("proj_code", code)
            putExtra("proj_data", data)
        }
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) startForegroundService(intent) else if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) startForegroundService(intent) else startService(intent)
        // ✅ ลบ moveTaskToBack(true) ออก → Android 14+ บล็อก MediaProjection ถ้าแอปไม่อยู่หน้าตา
    }

    private fun setLoading(on: Boolean) {
        progressBar.visibility  = if (on) View.VISIBLE else View.GONE
        btnValidate.isEnabled   = !on
        if (on) layoutActions.visibility = View.GONE
    }

    private fun showStatus(msg: String, success: Boolean) {
        tvStatus.text = msg
        tvStatus.setTextColor(if (success) 0xFF00C853.toInt() else 0xFFFF5252.toInt())
        tvStatus.visibility = View.VISIBLE
    }
}
